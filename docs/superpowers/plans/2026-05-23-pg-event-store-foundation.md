# PG Event Store + Snapshot (Foundation) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Postgres-backed event store — append-only `event_log` plus transactionally-maintained snapshot tables (`position_state`, `offer_claims`) — that any future cold-start can read with one SELECT (no replay), with a deterministic `rebuild_from_log`. Built and tested **in isolation**; the daemon is NOT rewired (still on Axiom) until Plan 2.

**Architecture:** A new `modules/execution/event_store/` package. `PostgresEventStore.append(session, event)` writes the `event_log` row AND folds the event into both snapshot tables **inside the caller's session** (one txn → log and snapshot can never diverge). Snapshot maintenance reuses the existing pure `transition()` FSM (`registry_offers.py`) for `offer_claims` and the existing `PaperPositionLedger` delta math for `position_state`. Projections gain `from_snapshot()` constructors. `rebuild_snapshot_from_log()` folds the whole log to recompute snapshots (audit/repair). We own the JSONB schema, so events serialize with clean field names (no Axiom legacy-name baggage).

**Tech Stack:** Python 3.13, SQLAlchemy 2.0 async (asyncpg runtime / psycopg for alembic), Alembic, pytest (`asyncio_mode=auto`), testcontainers Postgres for integration, sqlite-in-memory for unit. `uv` for all commands. `mypy --strict` enforced.

**Scope guardrails (do NOT do here — deferred to Plan 2/3):**
- Do NOT touch `daemon.py` boot wiring, `replay_from_axiom`, `AxiomEventSink`, or any Axiom file.
- Do NOT add `ReservationIntent`/`ReservationFailed` events or use `RegistryState.PENDING` (Plan 3, A2).
- Do NOT add the `diagnostics` table (Plan 3).
- All new claims have a `venue_offer_id`; `offer_claims.venue_offer_id` is nullable in schema (for Plan 3) but always set here.

from __future__ import annotations

from decimal import Decimal
from typing import Any
from uuid import UUID

from bfx_funding_bot.modules.execution.events import (
    OrderFilled,
    ReservationClaimed,
    ReservationReleased,
)

# event_type string <-> domain class. Clean field names (we own this schema).
_TYPE_BY_CLASS: dict[type, str] = {
    ReservationClaimed: "RESERVATION_CLAIMED",
    OrderFilled: "ORDER_FILL",
    ReservationReleased: "RESERVATION_RELEASED",
}
_CLASS_BY_TYPE = {v: k for k, v in _TYPE_BY_CLASS.items()}


def event_type_of(event: object) -> str:
    try:
        return _TYPE_BY_CLASS[type(event)]
    except KeyError:
        raise ValueError(f"unserializable event type: {type(event).__name__}") from None


def serialize_event(event: object) -> dict[str, Any]:
    """Domain event -> JSON-safe payload dict. Decimal->str, UUID->str."""
    etype = event_type_of(event)
    out: dict[str, Any] = {}
    for field, value in vars_of(event).items():
        if isinstance(value, Decimal):
            out[field] = str(value)
        elif isinstance(value, UUID):
            out[field] = str(value)
        else:
            out[field] = value
    out["__event_type__"] = etype
    return out


def deserialize_event(event_type: str, payload: dict[str, Any]) -> object:
    cls = _CLASS_BY_TYPE.get(event_type)
    if cls is None:
        raise ValueError(f"unknown event_type: {event_type}")
    kwargs: dict[str, Any] = {}
    for field in _FIELDS[cls]:
        raw = payload.get(field)
        kwargs[field] = _coerce(cls, field, raw)
    return cls(**kwargs)


# --- helpers ---
import dataclasses


def vars_of(event: object) -> dict[str, Any]:
    return {f.name: getattr(event, f.name) for f in dataclasses.fields(event)}  # type: ignore[arg-type]


_FIELDS: dict[type, list[str]] = {
    cls: [f.name for f in dataclasses.fields(cls)]  # type: ignore[arg-type]
    for cls in _CLASS_BY_TYPE.values()
}

_DECIMAL_FIELDS = {"size_usdt"}
_UUID_FIELDS = {"signal_correlation_id"}


def _coerce(cls: type, field: str, raw: Any) -> Any:
    if raw is None:
        return None
    if field in _DECIMAL_FIELDS:
        return Decimal(str(raw))
    if field in _UUID_FIELDS:
        return UUID(str(raw))
    return raw
```

from __future__ import annotations

from typing import Any

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from bfx_funding_bot.modules.execution.event_store.serialization import (
    event_type_of,
    serialize_event,
)
from bfx_funding_bot.modules.execution.event_store.tables import EventLogRow

# Event types whose re-delivery must be deduped (idempotent fills/releases).
_DEDUP_TYPES = frozenset({"ORDER_FILL", "RESERVATION_RELEASED"})


class PostgresEventStore:
    """Append-only event log + (Task 5/6) transactional snapshot maintenance."""

    def __init__(self, *, deployment_environment: str) -> None:
        self._env = deployment_environment

    async def append(self, session: AsyncSession, event: object) -> bool:
        """Append one domain event in the caller's txn. Returns False if deduped (skipped).

        Caller commits (e.g. via session_scope). Does NOT commit here.
        """
        etype = event_type_of(event)
        account_id: str = getattr(event, "account_id")
        venue_offer_id: str | None = getattr(event, "venue_offer_id", None)
        venue_seq: int | None = getattr(event, "venue_seq", None)
        cid: int | None = getattr(event, "cid", None)
        occurred_at_ms: int = getattr(event, "occurred_at_ms") or 0

        if etype in _DEDUP_TYPES and await self._already_logged(
            session, account_id, etype, venue_offer_id, venue_seq
        ):
            return False

        payload: dict[str, Any] = serialize_event(event)
        session.add(EventLogRow(
            account_id=account_id,
            deployment_environment=self._env,
            event_type=etype,
            cid=cid,
            venue_offer_id=venue_offer_id,
            venue_seq=venue_seq,
            payload=payload,
            occurred_at_ms=occurred_at_ms,
        ))
        # Snapshot maintenance is added in Task 5 (offer_claims) and Task 6 (position_state).
        return True

    async def _already_logged(
        self, session: AsyncSession, account_id: str, event_type: str,
        venue_offer_id: str | None, venue_seq: int | None,
    ) -> bool:
        stmt = (
            select(EventLogRow.event_seq)
            .where(
                EventLogRow.account_id == account_id,
                EventLogRow.deployment_environment == self._env,
                EventLogRow.event_type == event_type,
                EventLogRow.venue_offer_id == venue_offer_id,
                EventLogRow.venue_seq == venue_seq,
            )
            .limit(1)
        )
        return (await session.execute(stmt)).first() is not None
```

        self, session: AsyncSession, etype: str, account_id: str,
        size_usdt: Any, event_seq: int,
    ) -> None:
        from decimal import Decimal
        size = Decimal(str(size_usdt)) if size_usdt is not None else Decimal("0")
        ps = (await session.execute(

                PositionStateRow.account_id == account_id,
