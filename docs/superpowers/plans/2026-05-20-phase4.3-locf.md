```python
from dataclasses import dataclass


@dataclass(frozen=True)
class FilledCandle:
    """Wrapper for a candle slot after LOCF reindexing.

    candle is None when the slot is beyond max_gap_hours from any source candle
    (hard tier — caller must skip signal emit and emit DEGRADED instead).
    """
    candle: FundingCandle | None
    is_stale: bool
    stale_seconds: int


def reindex_and_ffill(
    candles: list[FundingCandle],
    ref_mts: int,
    freq_ms: int = 3_600_000,
    max_gap_hours: int = 2,
) -> list[FilledCandle]:
    """Reindex sparse candle list to a full hourly grid ending at ref_mts.

    Forward-fills missing slots up to max_gap_hours of staleness. Beyond budget,
    slots remain (candle=None, is_stale=True, stale_seconds=(ref - source)/1000).

    Invariants:
    - len(output) == n hourly slots from candles[0].mts to ref_mts (or all None
      slots back to ref_mts - lookback if candles is empty — see Test 1.5)
    - output[-1] aligns to ref_mts (last slot within freq_ms tolerance)
    - For 100% dense input on freq_ms grid: 1-to-1 wrap, no ffill applied
    - Deterministic: same input → same output across N invocations
    """
    if not candles:
        return []
    sorted_candles = sorted(candles, key=lambda c: c.mts)
    if sorted_candles[-1].mts > ref_mts:
        # Defensive: trim future candles (shouldn't happen but be explicit)
        sorted_candles = [c for c in sorted_candles if c.mts <= ref_mts]
        if not sorted_candles:
            return []

    start_mts = sorted_candles[0].mts
    n_slots = ((ref_mts - start_mts) // freq_ms) + 1
    max_gap_ms = max_gap_hours * 3_600_000

    output: list[FilledCandle] = []
    source_idx = 0
    source = sorted_candles[0]

    for slot in range(n_slots):
        slot_mts = start_mts + slot * freq_ms
        # Advance source pointer to the latest candle with mts <= slot_mts
        while (
            source_idx + 1 < len(sorted_candles)
            and sorted_candles[source_idx + 1].mts <= slot_mts
        ):
            source_idx += 1
            source = sorted_candles[source_idx]

        if source.mts == slot_mts:
            output.append(FilledCandle(candle=source, is_stale=False, stale_seconds=0))
            continue

        age_ms = slot_mts - source.mts
        if age_ms <= max_gap_ms:
            output.append(
                FilledCandle(
                    candle=source,
                    is_stale=True,
                    stale_seconds=age_ms // 1000,
                )
            )
        else:
            output.append(
                FilledCandle(
                    candle=None,
                    is_stale=True,
                    stale_seconds=age_ms // 1000,
                )
            )
    return output
```

async def run_one_wfo(
    session_factory,
    symbol: str,
    period_agg: str,
    strategy_cls,
    budget_hours: int,
    train_months: int = 3,
    test_months: int = 1,
    walk_months: int = 1,
) -> dict:
    """Run one (cell × strategy × budget) WFO with LOCF preprocessing.

    Returns dict with eligible, wins, win_pct, margin, health_pct, qualifies.
    """
    # Load all candles in post-2022-01-01 window
    start_mts = 1_640_995_200_000  # 2022-01-01 UTC
    # end_mts = now or sync Phase 3b cutoff

    async with session_factory() as session:
        candles = await get_candles_in_range(
            session=session,
            symbol=symbol,
            period_agg=period_agg,
            timeframe="1h",
            start_mts=start_mts,
            end_mts=int(__import__("time").time() * 1000),
        )

    # Iterate WFO windows
    windows = walk_forward_windows(
        candles=candles,
        train_months=train_months,
        test_months=test_months,
        walk_months=walk_months,
    )
    wins = 0
    eligible = 0
    healthy = 0
    margins = []
    for window in windows:
        # Apply LOCF to test window candles
        filled = reindex_and_ffill(
            candles=window.test_candles,
            ref_mts=window.test_end_mts,
            max_gap_hours=budget_hours,
        )
        unwrapped = [f.candle for f in filled if f.candle is not None]

        if len(unwrapped) < 10:  # too thin
            continue

        result = await run_backtest(
            candles=unwrapped,
            strategy_cls=strategy_cls,
            train_candles=window.train_candles,
        )
        eligible += 1
        if result.net_monthly_return_pct > 0:
            wins += 1
            margins.append(result.net_monthly_return_pct)
        if result.health_check_ok:
            healthy += 1

    win_pct = wins / eligible if eligible > 0 else 0.0
    margin = (sum(margins) / len(margins)) if margins else 0.0
    health_pct = healthy / eligible if eligible > 0 else 0.0
    qualifies = win_pct >= 0.60 and margin >= 0.05 and health_pct >= 0.80

    return {
        "cell": f"{symbol}_{period_agg}",
        "strategy": strategy_cls.__name__,
        "budget_hours": budget_hours,
        "eligible": eligible,
        "wins": wins,
        "win_pct": win_pct,
        "margin": margin,
        "health_pct": health_pct,
        "qualifies": qualifies,
    }


def _emit_markdown(results: list[dict], output_path: Path) -> None:
    """Write GREEN/YELLOW/RED matrix + per-pair results to markdown."""
    lines = [
        "# Phase 4.3 LOCF Backtest Sensitivity Sweep Results\n",
        f"\n**Run date**: {__import__('datetime').datetime.utcnow().isoformat()}\n",
        "\n## Per-pair × budget results\n",
        "\n| Cell | Strategy | Budget | Eligible | Wins | Win % | Margin | Health % | Qualifies |",
        "|---|---|---|---|---|---|---|---|---|",
    ]
    for r in results:
        lines.append(
            f"| {r['cell']} | {r['strategy']} | {r['budget_hours']}h | "
            f"{r['eligible']} | {r['wins']} | {r['win_pct']:.2%} | "
            f"{r['margin']:.4f} | {r['health_pct']:.2%} | "
            f"{'yes' if r['qualifies'] else 'no'} |"
        )

    # Compute GREEN/YELLOW/RED verdict per (cell, strategy) pair
    lines.append("\n## Ship gate verdict per pair (Phase 3b ≥60% win-rate + ≥5% margin)\n")
    lines.append("\n| Cell | Strategy | 6h | 12h | 24h | Verdict | Action |")
    lines.append("|---|---|---|---|---|---|---|")

    pairs = {}
    for r in results:
        key = (r["cell"], r["strategy"])
        if key not in pairs:
            pairs[key] = {}
        pairs[key][r["budget_hours"]] = r["qualifies"]

    for (cell, strategy), budgets in pairs.items():
        q6 = budgets.get(6, False)
        q12 = budgets.get(12, False)
        q24 = budgets.get(24, False)
        if q12:
            verdict = "**GREEN**"
            action = "ship; cells.yaml budget stays at 12h"
        elif q24:
            verdict = "**YELLOW**"
            action = "raise cells.yaml budget for this pair to 24h; re-run G2 audit"
        else:
            verdict = "**RED**"
            action = "disqualify pair from 4.4 canary (LOCF semantic breaks strategy)"
        lines.append(
            f"| {cell} | {strategy} | "
            f"{'✓' if q6 else '✗'} | "
            f"{'✓' if q12 else '✗'} | "
            f"{'✓' if q24 else '✗'} | "
            f"{verdict} | {action} |"
        )

    output_path.write_text("\n".join(lines))
    print(f"Wrote {output_path}")


async def amain(args) -> None:
    engine = get_engine(args.database_url)
    session_factory = async_sessionmaker(engine, expire_on_commit=False)

    results = []

                print(f"  → qualifies={r['qualifies']}, win_pct={r['win_pct']:.2%}, margin={r['margin']:.4f}")

    # Write JSON + markdown
    args.output_path.with_suffix(".json").write_text(json.dumps(results, indent=2))
    _emit_markdown(results, args.output_path)

from bfx_funding_bot.external.axiom.client import AxiomClient


P30_CELLS_FOR_AUDIT = [

async def query_stale_signals(axiom: AxiomClient, cells: list[str], start: str, end: str) -> list[dict]:
    """F1 query: all is_stale=True signals in window."""
    cell_list = ", ".join([f'"{c}"' for c in cells])
    query = f"""
        ['bfx-funding-bot']
        | where _time >= datetime({start}) and _time <= datetime({end})
        | where event_type == "signal"
        | where ['payload.cell_key'] in ({cell_list})
        | where ['payload.is_stale'] == true
        | project _time, ['payload.cell_key'], ['payload.signal_score'],
                  ['payload.stale_seconds'], ['payload.candle_close']
    """
    return await axiom.query_apl(query)

        | where event_type == "signal"
        | where ['payload.cell_key'] in ({cell_list})
        | where ['payload.is_stale'] == false
    """
    return await axiom.query_apl(query)


async def query_sparseness_rate(axiom: AxiomClient, cells: list[str], start: str, end: str) -> list[dict]:
    """F3 query: stale_exceeded events per cell per day."""
    cell_list = ", ".join([f'"{c}"' for c in cells])

        | where ['payload.reason'] == "stale_exceeded"

    """
    return await axiom.query_apl(query)


async def join_with_decisions(axiom: AxiomClient, signals: list[dict], start: str, end: str) -> list[dict]:
    """For each signal, look up matching decision event for win/loss outcome.

    decision event (Phase 4.1 schema) has `pnl_proxy` and `cell_key` matching signal.
    Win = pnl_proxy > 0.
    """
    query = f"""
        ['bfx-funding-bot']
        | where _time >= datetime({start}) and _time <= datetime({end})
        | where event_type == "decision"
        | project _time, ['payload.cell_key'], ['payload.pnl_proxy'], ['payload.signal_score']
    """
    decisions = await axiom.query_apl(query)

    # Index decisions by cell_key + nearest signal_score (or _time)
    # Simple join: match by (_time, cell_key) within ±1min tolerance
    joined = []
    for sig in signals:
        match = next(
            (d for d in decisions
             if d.get("payload.cell_key") == sig.get("payload.cell_key")
             and abs(d.get("_time_unix", 0) - sig.get("_time_unix", 0)) < 60),
            None,
        )
        if match:
            sig["pnl_proxy"] = match.get("payload.pnl_proxy", 0.0)
            sig["is_win"] = sig["pnl_proxy"] > 0
            joined.append(sig)
    return joined


def compute_verdict(stale_joined: list[dict], cell_key: str) -> dict:
    """Apply Phase 3b qualification to stale subset for one cell."""
    cell_signals = [s for s in stale_joined if s.get("payload.cell_key") == cell_key]
    n = len(cell_signals)
    if n < MIN_SAMPLE_SIZE:
        return {
            "cell_key": cell_key,
            "n_stale": n,
            "verdict": "YELLOW",
            "rationale": f"sample size {n} < {MIN_SAMPLE_SIZE} — extend window",
        }
    wins = sum(1 for s in cell_signals if s["is_win"])
    win_pct = wins / n
    margin = sum(s["pnl_proxy"] for s in cell_signals if s["is_win"]) / n  # simplified
    qualifies = win_pct >= WIN_RATE_FLOOR and margin >= MARGIN_FLOOR

    verdict = "GREEN" if qualifies else "RED"
    return {
        "cell_key": cell_key,
        "n_stale": n,
        "wins": wins,
        "win_pct": win_pct,
        "margin": margin,
        "verdict": verdict,
    }


async def run_audit(
    axiom_client: AxiomClient,
    cells: list[str],
    window_start: str,
    window_end: str,
    output_path: Path,
) -> None:
    stale = await query_stale_signals(axiom_client, cells, window_start, window_end)
    fresh = await query_fresh_signals(axiom_client, cells, window_start, window_end)
    sparseness = await query_sparseness_rate(axiom_client, cells, window_start, window_end)

    stale_joined = await join_with_decisions(axiom_client, stale, window_start, window_end)

    # Verdict per cell
    verdicts = [compute_verdict(stale_joined, c) for c in cells]

    # Emit markdown
    lines = [
        f"# Phase 4.3 LOCF G2 Calibration Audit — {window_start} to {window_end}\n",
        "\n## Stale subset verdict per cell\n",
        "\n| Cell | n_stale | wins | win % | margin | Verdict |",
        "|---|---|---|---|---|---|",
    ]
    for v in verdicts:
        lines.append(
            f"| {v['cell_key']} | {v['n_stale']} | {v.get('wins', '-')} | "
            f"{v.get('win_pct', 0):.2%} | {v.get('margin', 0):.4f} | "
            f"**{v['verdict']}** |"
        )

    lines.append("\n## Sparseness rate per cell per day\n")
    lines.append("\n| Date | Cell | Stale count |")
    lines.append("|---|---|---|")
    for s in sparseness:
        lines.append(
            f"| {s.get('_time', '?')} | {s.get('payload.cell_key', '?')} | "
            f"{s.get('stale_count_', 0)} |"
        )

    output_path.write_text("\n".join(lines))
    print(f"Wrote G2 audit to {output_path}")
